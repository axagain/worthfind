``` sql
--- 分区表创建
DROP TABLE IF EXISTS dws_sales_forecast_summary_1d
;
CREATE TABLE IF NOT EXISTS dws_sales_forecast_summary_1d
(
    sku                         STRING COMMENT 'SKU'
    ,month_1_forecast           BIGINT COMMENT '未来30天预测销量'
    ,month_2_forecast           BIGINT COMMENT '未来60天预测销量'
    ,month_3_forecast           BIGINT COMMENT '未来90天预测销量'
    ,month_4_forecast           BIGINT COMMENT '未来120天预测销量'
    ,month_5_forecast           BIGINT COMMENT '未来150天预测销量'
    ,month_6_forecast           BIGINT COMMENT '未来180天预测销量'
    ,day_55_forecast            BIGINT COMMENT '未来55天预测销量'
    ,day_75_forecast            BIGINT COMMENT '未来75天预测销量'
    ,day_90_forecast            BIGINT COMMENT '未来90天预测销量'
    ,day_95_forecast            BIGINT COMMENT '未来95天预测销量'
    ,day_110_forecast           BIGINT COMMENT '未来110天预测销量'
    ,day_115_forecast           BIGINT COMMENT '未来115天预测销量'
    ,day_135_forecast           BIGINT COMMENT '未来135天预测销量'
)
COMMENT '全渠道库存看板销量预测汇总快照表'
PARTITIONED BY (pt STRING COMMENT '业务日期 yyyyMMdd')
STORED AS ALIORC
TBLPROPERTIES ('comment' = '全渠道库存看板销量预测汇总快照表','auto.purge' = 'true')
LIFECYCLE 36500
```

``` sql
--- 非分区表创建

DROP TABLE IF EXISTS dws_sales_forecast_allocation_1d
;
CREATE TABLE IF NOT EXISTS dws_sales_forecast_allocation_1d
(
    sku                           STRING COMMENT '组织ID'
    ,channel_name                 STRING COMMENT '渠道'
    ,station_name                 STRING COMMENT '国家'
    ,daily_date                   DATETIME COMMENT '日期'
    ,daily_forecast_sale_quantity BIGINT COMMENT '预测销量'
)
COMMENT '销量预测分摊表'
LIFECYCLE 36500
;
```
